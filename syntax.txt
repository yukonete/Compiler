Current syntax:

// Constant declaration
const <identifier> [:<type>] = <value>;

// Variable declaration
var   <identifier> [:<type>] [=<value>];

// Type declaration
type  <identifier> = <type>;

// Function declaration
fn <identifier>([arg1, arg2, ...]) [-> <type>] {
	body;
}

// For compiler this
fn identifier() -> (identifier: type) {}
// Has the same meaning as this
fn identifier() -> type {}

// Struct
type Struct = struct {
	field: type;
	...
};

// Array
type IntArray = [100]int;

// Pointer
type IntPtr = *int;

// Function (pointer) type
type Fn = fn([arg1, arg2, ...]) -> <type>;
// This is allowed as well
type Fn = fn() -> (identifier: type);

// If
if expression {
} else if {
} else {
}

// While
while expression {
	statements
}

// Pointers
var b: int;
var a: *int;
a = &b;
var q: int = a.*;
