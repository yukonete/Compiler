Current syntax:

// Constant declaration
const <identifier> [:<type>] = <value>;

// Variable declaration
var   <identifier> [:<type>] [=<value>];

// Type declaration
type  <identifier> = <type>;

// Function declaration
fn <identifier>([arg1, arg2, ...]) -> <type> {
	
}

// Struct
type Struct = struct {
	field: type;
	...
};

// Array
type IntArray = [100]int;

// Pointer
type IntPtr = *int;

// Function (pointer) type
type Fn = fn([arg1, arg2, ...]) -> <type>;

// If
if expression statement
else statement

// While
while expression statement

// Pointers
// * applied to type means pointer to that type
// & applied to variable takes adress of it
// * applied to pointer dereferences that pointer
var a: *int;
a = &b;
var q: int = *b;
